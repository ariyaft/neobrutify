Quick Start
// 1. Include the script
<script src="brutalist.js"></script>

// 2. Generate an image
const svg = brutalist({
  width:  800,
  height: 600,
  palette: 'gumroad',
});

// 3. Add it to the page
document.body.appendChild(svg);
Full Options
const svg = brutalist({
  width:        1200,       // Image width (px)
  height:       630,        // Image height (px)
  palette:      'sunset',   // Palette name or custom object
  borderWidth:  3,          // Stroke width (px)
  shadowOffset: 6,          // Hard shadow offset (px)
  density:      2,          // 1 (sparse) → 5 (dense)
  seed:         'hello',    // Seed (string, number, or null)
  showText:     true,       // Show text overlay
  textContent:  'BRUT',    // Text to display
});
Export
// Download as PNG (2x resolution)
brutalist.downloadPNG({ width: 800, height: 600 }, 'image.png', 2);

// Download as SVG
brutalist.downloadSVG(svg, 'image.svg');

// Get SVG string
const svgString = brutalist.toSVGString(svg);

// Get data URL
const dataUrl = brutalist.toSVGDataURL(svg);

// Render to canvas
brutalist.toCanvas({ width: 800, height: 600 }).then(canvas => {
  document.body.appendChild(canvas);
});
